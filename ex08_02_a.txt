// 아두이노: 수신된 정수로 LED 점멸 제어

#define LED 6

void setup() {
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
}

void loop() { }

void serialEvent() {		// 이벤트함수: 수신값이 있으면 자동 호출
  int value = Serial.read();	// 수신값 읽기
  if (value == 1) {    		// 수신값이 1인 경우
    digitalWrite(LED, 1);
  } else if (value == 0) {  	// 수신값이 0인 경우
    digitalWrite(LED, 0);
  }
}
