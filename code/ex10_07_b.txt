// 프로세싱 (인터럽트 방식): 수신값 2개로 원 이동, 마우스의 상하좌우 좌표값 2개를 송신

import processing.serial.*;			// 시리얼 통신 라이브러리 불러오기
Serial port;					// Serial 객체 선언
float value1 = 0, value2 = 0;
int r = 0, g = 0;

void setup() {
  size(500, 500);
  port = new Serial(this, "COM3", 9600);	// Serial 객체 생성(포트, 통신속도 설정)
}

void draw() {
  // 페이딩 효과
  fill(0, 10);					// 투명도 부여
  noStroke();					// 외곽선 제거
  rect(0, 0, width, height);			// 사각형 크기를 실행창과 동일하게 설정

  r = int(map(mouseX, 0, width, 0, 255));	// RGB LED의 Red값
  g = int(map(mouseY, 0, height, 0, 255));	// RGB LED의 Green값

  fill(r, g, 255);					// 원의 색상에 적용
  ellipse(value1, value2, 50, 50);		// 수신값 2개로 원을 상하 및 좌우 이동
}

void serialEvent(Serial port) {			// 시리얼 버퍼에 값이 들어오면 자동 호출
  String str = port.readStringUntil('\n');		// 수신: 종료문자('\n')까지 수신 문자열 읽기
  if (str != null) {				// 수신 문자열이 있을 때
    String s = trim(str);			// 문자열 앞뒤의 공백 제거
    String[] value = split(s, ",");		// 콤마로 구분된 문자열을 배열에 저장
    port.write(r + " " + g + "\n");		// 송신: 복수 값을 하나의 문자열로 전송

    if (value.length != 2) return;		// 통신시작신호('A')인 경우 함수 벗어나기
    value1 = map(float(value[0]), 0, 1023, 0, width);	// value[0]는 실수로 변환→원의 x좌표
    value2 = map(float(value[1]), 0, 1023, 0, height);	// value[1]은 실수로 변환→원의 y좌표
  }
}
