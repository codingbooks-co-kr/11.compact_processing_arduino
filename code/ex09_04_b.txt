// 프로세싱: 가변저항값 수신, 원 좌우 이동

import processing.serial.*;
Serial port;
int value = 0;

void setup() {
  size(500, 200);
  port = new Serial(this, "COM3", 9600);
}

void draw() {
  background(0);
  fill(255, 255, 0);
  textSize(30);
  text("Variable R: " + nf(value, 4), 20, 30);

  float x = map(value, 0, 1023, 0, width);
  ellipse(x, height/2, 80, 80);
}

// 수신값이 있으면 자동 호출
void serialEvent(Serial port) {
  String str = port.readStringUntil('\n');
  if (str != null) {
    String s = trim(str);
    value = int(s);
  }
}
