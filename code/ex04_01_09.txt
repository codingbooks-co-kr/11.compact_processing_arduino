// 방향키로 원을 상하좌우로 이동

int x = 150, y = 150;

void setup() {
  size(300, 300);
}

void draw() {
  background(0);
  if (keyPressed) {		// 중첩 if문: ❶키를 누른 경우 ❷어떤 키를 눌렀는지 판단
    if (keyCode == LEFT) {			// 왼쪽 방향 키
      x--;
    } else if (keyCode == RIGHT) {		// 오른쪽 방향 키
      x++;
    } else if (keyCode == UP) {		// 위쪽 방향 키
      y--;
    } else if (keyCode == DOWN) {		// 아래쪽 방향 키
      y++;
    }
  }
  ellipse(x, y, 100, 100);
}
