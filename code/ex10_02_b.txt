// 프로세싱 (인터럽트 방식): 마우스의 좌우 및 상하 움직임값 2개를 송신

import processing.serial.*;			// 시리얼 통신 라이브러리 불러오기
Serial port;					// Serial 객체 선언
int value1 = 0, value2 = 0;

void setup() {
  size(500, 500);
  port = new Serial(this, "COM3", 9600);	// Serial 객체 생성(포트, 통신속도 설정)
}

void draw() {
  // 페이딩 효과
  fill(0, 10);					// 투명도 부여
  noStroke();					// 외곽선 제거
  rect(0, 0, width, height);			// 사각형 크기를 실행창과 동일하게 설정

  value1 = int(map(mouseX, 0, width, 0, 255));	// 좌우 이동: RGB LED의 Red값
  value2 = int(map(mouseY, 0, height, 0, 255));	// 상하 이동: RGB LED의 Green값

  fill(value1, value2, 255);			// 원의 색상에 적용
  ellipse(mouseX, mouseY, 50, 50);		// 마우스를 따라 움직이는 원
}

void serialEvent(Serial port) {			// 이벤트함수: 수신값이 있으면 자동 호출
  int signal = port.read(); 			// 수신 (첫 실행 시 'A' 수신, 그후 'B' 수신)
  port.write(value1 + " " + value2 + "\n");	// 복수 값을 하나의 문자열로 묶어 송신
}
