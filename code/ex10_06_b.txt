// 프로세싱: 인터럽트 방식
// 마우스 클릭값 송신, 푸시버튼 클릭값 수신

import processing.serial.*;
Serial port;
int value1 = 0, value2 = 0;

void setup() {
  size(500, 500);
  port = new Serial(this, "COM3", 9600);
}

void draw() {
  background(0);
  textSize(30);
  if (mousePressed) {
    fill(255, 0, 0);
    text("LED ON", 20, 40);
    value1 = 1;		// 송신값
  } else {
    fill(255, 255, 0);
    text("LED OFF", 20, 40);
    value1 = 0;		// 송신값
  }

  if (value2 == 0) {		// 수신값
    fill(255, 0, 0);
  } else {
    fill(255, 255, 0);
  }
  ellipse(width/2, height/2, 300, 300);
}

// 수신값이 있으면 자동 호출
void serialEvent(Serial port) {
  value2 = port.read();		// 수신
  port.write(value1);		// 송신
}
