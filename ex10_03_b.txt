// 프로세싱: 인터럽트 방식
// 가변저항값 수신

import processing.serial.*;
Serial port;
int value = 0;

void setup( ) {
  size(500, 200);
  port = new Serial(this, "COM3", 9600);
}

void draw() {
  background(0);
  fill(255, 255, 0);
  textSize(30);
  text("Variable R: " + nf(value, 3), 20, 30);

  float x = map(value, 0, 255, 0, width);
  ellipse(x, height/2, 80, 80);
}

// 수신값이 있으면 자동 호출
void serialEvent(Serial port) {
  value = port.read();	// 수신
  port.write('B');	// 송신
}
