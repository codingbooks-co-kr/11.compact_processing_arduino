// 푸시버튼을 누르는 동안 LED 점멸 (함수유형2)

#define BUTTON 2
#define LED 6

void setup() {
  pinMode(BUTTON, INPUT);
  pinMode(LED, OUTPUT);
}

void loop() {
  int value = digitalRead(BUTTON);		// 버튼값 읽기
  if (value == 0) {
    blink(200);					// 함수 호출
  }
}

void blink(int time) {				// 함수 유형2
  digitalWrite(LED, 1);
  delay(time);
  digitalWrite(LED, 0);
  delay(time);
}
