// 프로세싱: 인터럽트 방식
// 마우스 위치값 송신, 가변저항값 수신

import processing.serial.*;
Serial port;
int value1 = 0, value2 = 0;

void setup() {
  size(500, 500);
  port = new Serial(this, "COM3", 9600);
}

void draw() {
  value1 = int(map(mouseX, 0, width, 0, 255));
  background(value1);

  float size = map(value2, 0, 255, 0, width);
  fill(255, 255, 0);
  ellipse(width/2, height/2, size, size);
}

// 수신값이 있으면 자동 호출
void serialEvent(Serial port) {
  value2 = port.read();		// 수신
  port.write(value1);		// 송신
}
