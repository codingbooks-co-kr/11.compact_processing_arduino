// 랜덤하지만 방향성을 가진 1000개의 점

int num = 1000;
float[] x = new float[num];
float[] y = new float[num];

void setup() {
  size(300, 300);
  for (int i = 0; i < num; i++) {	// 배열 요소의 초깃값 설정
    x[i] = random(width);		// x좌표 초기 위치 랜덤 설정
    y[i] = random(height);		// y좌표 초기 위치 랜덤 설정
  }
}

void draw() {
  // 페이딩 효과
  fill(0, 10);				// 투명도 부여
    noStroke();				// 외곽선 제거
    rect(0, 0, width, height);		// 사각형 크기를 실행창과 동일하게 설정

  stroke(255);
  for (int i = 0; i < num; i++) {
    point(x[i], y[i]);
    x[i] += random(1.0);		// x좌표 랜덤값(0.0∼1.0) 누적, 방향성: 오른쪽
    y[i] += random(1.0);		// y좌표 랜덤값(0.0∼1.0) 누적, 방향성: 아랫쪽
  }
}
