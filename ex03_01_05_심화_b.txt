// 덧셈 연산자인 ‘+’를 사용하여 문자열 연결

int i = 12;
float f = 12.345;
char c = 'A';
String s = "Processing!";
println("int i = " + i);
println("float f = " + f);
println("char c = " + c);
println("String s = " + s);
println("Hello " + "Processing!");
println(12 + 12.34);
println(12 + 12.34 + " = Sum");	// 연산+연결
println("Sum = " + 12 + 12.34);	// 연결
