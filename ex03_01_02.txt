// 다양한 크기의 원 그리기

int x = 100, y = 100, d = 80;		// 정수로 초기화

size(300, 300);
noFill();
ellipse(x%30, y%30, d%30, d%30);		// 나눈 나머지
ellipse(x/2, y/2, d/2, d/2);			// 나눗셈
ellipse(x-30, y-30, d-30, d-30);		// 뺄셈
ellipse(x, y, d, d);
ellipse(x+30, y+30, d+30, d+30);		// 덧셈
ellipse(x*2, y*2, d*2, d*2);			// 곱셈
